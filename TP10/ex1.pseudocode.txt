def factorielle(nombre : entier) retourne entier
debut
    si (nombre = 1 or nombre = 0) alors
        retourne 1
    sinon
        retourne nombre * factorielle(nombre-1)
fin

fonction combiDirecte(n : entier, k : entier) retourne entier
debut
    valeur : entier

    valeur <- factorielle(n) / (factorielle(k) * factorielle(n - k))
    retourne valeur
fin

fonction combiRecursive(n : entier, k : entier) retourne entier
debut
    si (k = 0 ou n = k) alors
        retourne 1
    sinon 
        retourne combiRecursive(n-1, k) + combiRecursive(n-1, k-1)
    finsi
fin

procedure saisieEntier(message : chaine, msgErreur : chaine, borneMin : entier, borneMax : entier, MinInclus : booleen, MaxInclus : booleen ; valeur : entier)
debut
    # Demande a l'utilisateur de saisir une valeur entiere.

    avec valeur : entier
    
    afficher(message)
    saisir valeur

    verifEntier(valeur, msgErreur, borneMin, borneMax, MinInclus, MaxInclus)

    retourne valeur
fin saisieEntier

procedure verifEntier(valeur, msgErreur, borneMin, borneMax, MinInclus, MaxInclus)
debut
    # Verifie si un reel est bien dans un intervale donne.

    si MinInclus et MaxInclus alors
        tant que valeur < borneMin ou valeur > borneMax faire
            afficher(msgErreur)
            saisir valeur
    
    sinonsi non MinInclus et MaxInclus alors
        tant que valeur <= borneMin ou valeur > borneMax faire
            afficher(msgErreur)
            saisir valeur
    
    sinonsi MinInclus et non MaxInclus alors
        tant que valeur < borneMin ou valeur >= borneMax faire
            afficher(msgErreur)
            saisir valeur
    
    sinon alors
        tant que valeur <= borneMin or valeur >= borneMax faire
            afficher(msgErreur)
            saisir valeur
fin verifEntier
