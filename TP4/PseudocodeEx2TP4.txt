procedure verificationEtudiant(message : chaine ; etudiantStr : chaine)
début
	tant que etudiantStr != "y" et etudiantStr != "n" faire
		Afficher(message)
		Saisir etudiantStr
	finfaire
fin

procedure conversionEtudiant(etudiantStr : chaine ; etudiant : booleen)
début
	si etudiantStr = "y" faire
		etudiant <- True
	sinon
		etudiant <- False
fin

procedure saisieEntier(affichage : chaine, msgErreur : chaine ; val: entier)
début
	Afficher(affichage)
	Saisir val

	verificationAge(val, msgErreur)
fin

procedure saisieEtudiant(affichage : chaine, msgErreur : chaine; EtudiantStr : chaine, etudiant);
début
	Afficher(affichage)
	saisir EtudiantStr

	verificationEtudiant(msgErreur, EtudiantStr)
	conversionEtudiant(EtudiantStr, etudiant)
